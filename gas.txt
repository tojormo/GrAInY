// ==========================================
// 設定項目
// ==========================================
// 1. 画像を保存する Google ドライブのフォルダID（URLの folders/ 以降の文字列）
const FOLDER_ID = "1iQuif-399e-eomRuBGZDmOvkhl7MGXtA";

// 2. フロントアプリと共通で使うセキュリティ用パスワード（任意の英数字）
const SECRET_KEY = "tojormo"; 

// 3. 対象のスプレッドシートID（URLから取得した文字列）
const SPREADSHEET_ID = "1ClN1L4zZF78I9DcXmygh47TTvmbrEbHKwdj5Q3ySaBk";

// ==========================================
// メイン処理（POSTリクエスト受信時）
// ==========================================
function doPost(e) {
  const createResponse = (status, message, extraData = {}) => {
    return ContentService.createTextOutput(JSON.stringify({ status, message, ...extraData }))
      .setMimeType(ContentService.MimeType.JSON);
  };

  try {
    if (!e || !e.postData || !e.postData.contents) {
      return createResponse("error", "リクエストデータが空です");
    }
    
    const data = JSON.parse(e.postData.contents);

    // 1. パスワード検証
    if (data.secretKey !== SECRET_KEY) {
      return createResponse("error", "認証エラー: セキュリティキーが一致しません");
    }

    // ★ userName と imgHeight（高さ）を追加で受け取る
    const { userName, fieldId, lat, lng, imgEar, imgRoot, imgHeight, imgView } = data;
    if (!fieldId || !lat || !lng) {
      return createResponse("error", "圃場IDまたは位置情報が不足しています");
    }
    // ★ ユーザー名の必須チェック
    if (!userName) {
      return createResponse("error", "ユーザー名が不足しています");
    }

    // 2. Googleドライブフォルダの取得
    const folder = DriveApp.getFolderById(FOLDER_ID);

    // 3. 画像のデコード＆ドライブ保存用関数（軽量・高速化）
    const saveImageToDrive = (base64Data, categoryLabel) => {
      if (!base64Data || typeof base64Data !== 'string') return "画像なし";

      try {
        // Base64のヘッダーを取り除く
        const base64Clean = base64Data.replace(/^data:image\/\w+;base64,/, "");
        const decodedBytes = Utilities.base64Decode(base64Clean);
        
        // 命名規則: 圃場ID_写真カテゴリ.jpg
        const fileName = `${fieldId}_${categoryLabel}.jpg`;
        const blob = Utilities.newBlob(decodedBytes, "image/jpeg", fileName);
        
        // ドライブにファイル保存（※個別のsetSharingは負荷軽減のため省略）
        const file = folder.createFile(blob);
        return file.getUrl();
      } catch (imgErr) {
        Logger.log(`画像保存エラー [${categoryLabel}]: ` + imgErr.toString());
        return "保存失敗";
      }
    };

    // 4. 画像4枚を順番に保存（★「高さ」を追加）
    const urlEar    = saveImageToDrive(imgEar,    "稲穂");
    const urlRoot   = saveImageToDrive(imgRoot,   "根本");
    const urlHeight = saveImageToDrive(imgHeight, "高さ");   // ★ 追加
    const urlView   = saveImageToDrive(imgView,   "圃場前景");

    // 5. スプレッドシートへデータ書き込み
    let sheet;
    if (SPREADSHEET_ID && SPREADSHEET_ID.trim() !== "") {
      sheet = SpreadsheetApp.openById(SPREADSHEET_ID).getActiveSheet();
    } else {
      sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    }

    const timestamp = Utilities.formatDate(new Date(), "Asia/Tokyo", "yyyy/MM/dd HH:mm:ss");

    // ★ B列にユーザー名、G列に高さ画像URLを追加
    //   新しい列構成:
    //   A:日時 B:ユーザー名 C:圃場ID D:緯度 E:経度
    //   F:稲穂URL G:根本URL H:高さURL I:圃場前景URL
    sheet.appendRow([
      timestamp, // A列: 日時
      userName,  // B列: ユーザー名 ★追加
      fieldId,   // C列: 圃場ID
      lat,       // D列: 緯度
      lng,       // E列: 経度
      urlEar,    // F列: 稲穂画像URL
      urlRoot,   // G列: 根本画像URL
      urlHeight, // H列: 高さ画像URL ★追加
      urlView    // I列: 圃場前景画像URL
    ]);

    return createResponse("success", "データが正常に保存されました");

  } catch (error) {
    return createResponse("error", "サーバーエラーが発生しました: " + error.toString());
  }
}
